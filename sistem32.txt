void god(){ // efecto de sonido good
  b = false;
  tone(D5, 900);
  delay(150);
  tone(D5, 800);
  delay(150);
  noTone(5);
  b = true;
}
void damage(){ // efecto sonido daño

}
void iper(int x, int y){ // icono personaje
  l.fillCircle(x + 10, y + 10, 5, WHITE);

}
void ifiles(int x, int y){ // icono archivos

}

void setup() {
  randomSeed(analogRead(A0) + micros());
  l.begin(SSD1306_SWITCHCAPVCC, 0x3c);
  l.display();
  Wire.begin();
  l.clearDisplay();
  l.setTextColor(WHITE);
  l.setTextSize(2);
  mpu.initialize();
  Serial.begin(9600);
  pinMode(D5, OUTPUT);
  pinMode(D6, INPUT_PULLUP);
  // put your setup code here, to run once:
  
}
void music(){

  if ( millis() - ta > times[ia]) {

    ia++;

    if (ia >= 52) ia = 0;

    tone(D5, notes[ia]);

    ta = millis();
  }
}
void cur(int x, int y){
  l.fillTriangle(x -5, y -5, x +5, y -5, x, y, WHITE); 
}

 if(e == 1){
